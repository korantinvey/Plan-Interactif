/* Le mode administration (§ 10) : fiche et masquage d'une zone, libellés placés à la main. Suite de `_ordre-fiche.html`, dans le même script. */ /* ============================================================ 10. Mode administration — le branchement Il n'existe que dans la page dédiée, et seulement une fois l'utilisateur authentifié. Son état, `ADMIN`, et le retrait de ses commandes de la page publique vivent dans `modules/mode-admin.mjs` ; le profil du compte, `PROFIL_ADMIN`, dans `modules/acces-admin.mjs`, qui le lit. Le code soudé les lit par accesseur, comme avant par leur nom. Le reste est d'exploitant, dans des modules que seul `plan-admin.mjs` embarque : l'ouverture du mode et la bande de l'outil (`modules/bande-admin.mjs`), la fiche d'une zone, son masquage et l'écriture des colonnes de l'événement (`modules/fiche-zone.mjs`), l'outil qui place un libellé à la main (`modules/placement-libelles.mjs`). Ce que le visiteur reçoit de ce dernier — le placement relu au dessin des noms — est dans `modules/libelle-place.mjs`. Ils se branchent ici, à la place que leur code tenait, et ce que le code soudé tient encore leur est confié — par des lecteurs pour ce qui change ou se déclare plus loin dans le script. ============================================================ */ /* @admin — l'ouverture du mode administration, que seule la porte authentifiée appelle */ /* `construitPanneau` se déclare plus loin, dans `_pile.html` : elle se confie par un appel différé, lu au moment d'ouvrir le mode. */ brancheBandeAdmin({ ouvreReglages, monte: () => MONTE, dessineDessins, construitPanneau: () => construitPanneau(), poseTonDeLaBarre, }); /* @fin-admin */ /* @admin — la fiche d'une zone, qu'on ne remplit qu'en administration */ /* Les types de zone et la clé d'un emplacement ajouté se déclarent plus loin, dans `_dessin.html` et `_geometrie.html` : on les confie par un détour. */ brancheFicheZone({ conf: () => CONF, typesZone: () => TYPES_ZONE, typeZone: (z) => typeZone(z), cleAjout: (id) => cleAjout(id), annonce, enregistreConf, nomsAnglaisDesZones, rechAjout, libelles, liste, cartouchePoi, majPaletteGeo, ouvre, rangeConferences, }); /* @fin-admin */ /* Le placement réglé à la main, que le dessin des noms relit chez le visiteur. */ brancheLibellePlace({ conf: () => CONF, ancre, place }); /* @admin — le placement des libellés à la main */ branchePlacementLibelles({ svg, conf: () => CONF, enregistreConf, libelles, versPlan, calqueActif: () => calqueActif, activeCalque, sorteGeo: () => SORTE_GEO, modeGeometrie, }); /* @fin-admin */
