/* ============================================================ Classeur — écrire un vrai fichier Excel, sans bibliothèque L'organisateur veut rendre des comptes à ses exposants, et il le fait dans un tableur : c'est là qu'il trie, filtre, et colle la ligne d'un exposant dans le courriel qu'il lui envoie. Un CSV l'aurait servi à moitié — Excel y devine le séparateur de travers une fois sur deux, selon la langue du poste, et un nombre s'y lit comme du texte. Pourquoi rien n'est chargé d'ailleurs : la console est servie telle quelle par Cloudflare, sans étape de construction ; une dépendance ajoutée ici serait un script tiers de plus dans une page d'administration. Or un `.xlsx` est une archive ZIP de quelques fichiers XML, et une archive sans compression n'a besoin que d'un CRC32. Cela tient en deux cents lignes, qu'on relit, plutôt qu'en trois cents kilo-octets qu'on ne relit pas. Ce qu'il sait faire, et rien de plus : une feuille, une ligne d'en-têtes en gras et figée, des colonnes larges à la demande, du texte et des nombres. ============================================================ */ /** * CRC32, table précalculée. * * Le calcul bit à bit suffirait, mais l'archive fait le mégaoctet sur un gros * salon : huit millions de tours de boucle se sentent sur un poste modeste, et * la table les divise par huit pour deux cent cinquante-six entiers. */ const CRC_TABLE = (() => { const t = new Uint32Array(256); for (let i = 0; i < 256; i++){ let c = i; for (let k = 0; k < 8; k++) c = (c >>> 1) ^ (0xEDB88320 & -(c & 1)); t[i] = c >>> 0; } return t; })(); function crc32(u8){ let c = 0xFFFFFFFF; for (let i = 0; i < u8.length; i++) c = CRC_TABLE[(c ^ u8[i]) & 255] ^ (c >>> 8); return (c ^ 0xFFFFFFFF) >>> 0; } /** * Une archive ZIP, entrées rangées sans compression. * * Déflater économiserait les trois quarts du fichier — un classeur XML se * comprime très bien — mais demanderait `CompressionStream`, donc un chemin de * repli pour les navigateurs qui ne l'ont pas, donc deux codes à maintenir * pour un fichier qu'on ouvre et qu'on jette. Un mégaoctet enregistré depuis * la console de son propre salon ne coûte rien à personne. */ function archiveZip(fichiers){ const enc = new TextEncoder(); const d = new Date(); const heureDos = (d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1); const dateDos = ((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate(); const p16 = (t, n) => t.push(n & 255, (n >> 8) & 255); const p32 = (t, n) => t.push(n & 255, (n >>> 8) & 255, (n >>> 16) & 255, (n >>> 24) & 255); const morceaux = [], index = []; let pos = 0; for (const f of fichiers){ const nom = enc.encode(f.nom), data = enc.encode(f.texte); const somme = crc32(data); const tete = []; p32(tete, 0x04034b50); p16(tete, 20); // version minimale : rangé sans compression p16(tete, 0x0800); // noms en UTF-8 p16(tete, 0); // méthode : rangé p16(tete, heureDos); p16(tete, dateDos); p32(tete, somme); p32(tete, data.length); p32(tete, data.length); p16(tete, nom.length); p16(tete, 0); morceaux.push(new Uint8Array(tete), nom, data); index.push({ nom: nom, somme: somme, taille: data.length, pos: pos }); pos += tete.length + nom.length + data.length; } const centre = []; for (const e of index){ p32(centre, 0x02014b50); p16(centre, 20); p16(centre, 20); p16(centre, 0x0800); p16(centre, 0); p16(centre, heureDos); p16(centre, dateDos); p32(centre, e.somme); p32(centre, e.taille); p32(centre, e.taille); p16(centre, e.nom.length); p16(centre, 0); p16(centre, 0); p16(centre, 0); p16(centre, 0); p32(centre, 0); p32(centre, e.pos); centre.push(...e.nom); } const fin = []; p32(fin, 0x06054b50); p16(fin, 0); p16(fin, 0); p16(fin, index.length); p16(fin, index.length); p32(fin, centre.length); p32(fin, pos); p16(fin, 0); morceaux.push(new Uint8Array(centre), new Uint8Array(fin)); return new Blob(morceaux, { type: "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" }); } /* ------------------------------------------------------------------ Le classeur ------------------------------------------------------------------ */ /* Un nom d'exposant vient d'un fournisseur, et rien n'y interdit un caractère de commande. XML 1.0 les refuse : Excel n'ouvrirait pas le fichier, et l'erreur ne dirait pas pourquoi. On les retire avant d'échapper le reste. */ const texteXml = (v) => esc(String(v == null ? "" : v).replace(/[\x00-\x08\x0B\x0C\x0E-\x1F]/g, "")); /** « A », « B », … « AA » : la référence d'une colonne, à partir de 1. */ function colonneXl(n){ let s = ""; while (n > 0){ const r = (n - 1) % 26; s = String.fromCharCode(65 + r) + s; n = (n - r - 1) / 26; } return s; } /* Le squelette d'un classeur d'une feuille. Rien n'y varie que la feuille elle-même : ces quatre parties sont les mêmes d'un export à l'autre. */ const XL_DECL = ''; const XL_NS = "http://schemas.openxmlformats.org/spreadsheetml/2006/main"; const XL_NS_REL = "http://schemas.openxmlformats.org/officeDocument/2006/relationships"; const XL_PARTS = (feuille, titre) => [ { nom: "[Content_Types].xml", texte: XL_DECL + '' + '' + '' + '' + '' + '' + '' }, { nom: "_rels/.rels", texte: XL_DECL + '' + '' + '' }, { nom: "xl/workbook.xml", texte: XL_DECL + '' + '' + '' }, { nom: "xl/_rels/workbook.xml.rels", texte: XL_DECL + '' + '' + '' + '' }, /* Deux styles : ordinaire, et gras pour la ligne d'en-têtes. Les listes de polices, remplissages et bordures ne servent à rien ici, mais Excel refuse un classeur dont la feuille de styles n'en porte aucune. */ { nom: "xl/styles.xml", texte: XL_DECL + '' + '' + '' + '' + '' + '' + '' + '' + '' + '' + '' }, { nom: "xl/worksheets/sheet1.xml", texte: feuille }, ]; /** * Une feuille, à partir de colonnes et de lignes. * * colonnes : [{ titre, largeur }] — la largeur est en caractères * lignes : [[valeur, …]] — nombre ou texte, dans l'ordre des colonnes * * Les textes sont écrits en clair dans la cellule plutôt que renvoyés à une * table de chaînes partagées : celle-ci économiserait la place sur des valeurs * répétées, et il n'y en a pas ici — un nom d'exposant ne paraît qu'une fois. * * Une cellule vide est omise, pas écrite vide : c'est ce qu'attend le format, * et cela retire un tiers du fichier sur un salon où la plupart des canaux * sont à zéro. */ function feuilleXl(colonnes, lignes){ const derniere = colonneXl(colonnes.length); const cellule = (col, rang, v, style) => { if (v === "" || v == null) return ""; const r = ' r="' + colonneXl(col) + rang + '"' + (style ? ' s="' + style + '"' : ""); return typeof v === "number" && isFinite(v) ? "" + v + "" : '' + texteXml(v) + ""; }; const corps = [ "" + colonnes.map((c, i) => cellule(i + 1, 1, c.titre, 1)).join("") + "", ]; lignes.forEach((l, i) => { corps.push('' + colonnes.map((c, j) => cellule(j + 1, i + 2, l[j])).join("") + ""); }); const etendue = "A1:" + derniere + (lignes.length + 1); return XL_DECL + '' + '' + /* L'en-tête reste à l'écran : un salon aligne neuf cents lignes, et sans cela on ne sait plus au bout de trente ce que compte la colonne G. */ '' + '' + "" + '' + "" + colonnes.map((c, i) => '').join("") + "" + "" + corps.join("") + "" + // les filtres posés d'avance : c'est le premier geste de qui reçoit ce fichier '' + ""; } /** Le classeur complet, prêt à enregistrer. */ const classeurXl = (titre, colonnes, lignes) => archiveZip(XL_PARTS(feuilleXl(colonnes, lignes), titre)); /** * Proposer un fichier à l'enregistrement. * * L'adresse temporaire est révoquée après coup : sans cela le classeur — un * mégaoctet — resterait en mémoire tant que l'onglet vit, et la console reste * ouverte la journée. Le délai laisse au navigateur le temps de lire l'adresse * avant qu'elle ne disparaisse. */ function enregistreFichier(blob, nom){ const url = URL.createObjectURL(blob); const a = document.createElement("a"); a.href = url; a.download = nom; document.body.appendChild(a); a.click(); a.remove(); setTimeout(() => URL.revokeObjectURL(url), 20000); }
