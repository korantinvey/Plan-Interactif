/* ============================================================ 11 sexies. Le rappel avant une conférence ============================================================ */ /** * Le visiteur retient trois conférences le matin, puis passe sa journée dans * les allées. À quinze heures moins le quart il est à l'autre bout du hall, le * téléphone dans la poche, et la conférence qu'il avait notée commence sans * lui. Le parcours savait l'heure ; personne ne la lui a rappelée. * * Ce module est la moitié visible du réveil. L'autre moitié est un serveur, et * il faut dire pourquoi, parce que ce n'était pas le choix qu'on aurait voulu * faire : une page n'a aucun moyen de se réveiller elle-même. Ses minuteries * meurent avec l'onglet — gelé en quelques secondes dès qu'il passe en * arrière-plan sur un téléphone — le service de second plan est arrêté après * quelques secondes d'inactivité, et l'API qui aurait comblé ce manque * (*Notification Triggers*) n'a jamais quitté l'essai d'origine de Chrome. Un * rappel « local » ne serait donc parti qu'avec le plan ouvert sous les yeux, * c'est-à-dire au moment précis où il ne sert à rien. * * D'où la contrepartie, que le tiroir dit maintenant en toutes lettres : ce qui * est rappelé quitte l'appareil. Pas la liste — les stands retenus restent où * ils sont, et les conférences non plus ne partent pas d'elles-mêmes. Il faut * deux gestes : toucher l'interrupteur, puis accorder au navigateur * l'autorisation qu'il demande. Ce qui part alors est un abonnement — une * adresse opaque que le navigateur vient de tirer —, une heure, et la phrase à * afficher, déjà écrite. * * Déjà écrite, et c'est le point : la notification est composée ici, dans la * langue que le visiteur lisait, avec le nom que la conférence portait alors. * Le serveur la chiffre et la poste sans la relire, et le service de poussée du * navigateur ne peut pas la lire du tout. * * Ce qui ne marchera pas, et qu'il vaut mieux savoir. Sur iPhone, le Web Push * n'existe que pour un plan ajouté à l'écran d'accueil : dans un onglet Safari, * dans Chrome iOS, dans le navigateur intégré de LinkedIn ou d'Instagram, il * n'y a pas de `PushManager` du tout — l'interrupteur ne se montre alors pas, * et une ligne dit ce qu'il faudrait faire. Sur Android, les notifications de * Chrome lui-même peuvent être coupées au niveau du système, et les gestionnaires * de batterie de certains constructeurs retardent le réveil. Enfin un mode de * concentration retient le rappel sans que la page puisse rien y faire : le web * ne dispose pas du niveau d'interruption qu'une application native réclame. * Les horaires restent donc lisibles dans le tiroir, pour tous. */ /* ------------------------------------------------------------ Le réglage du salon ------------------------------------------------------------ */ /** * Un quart d'heure par défaut : le temps de finir une conversation sur un stand * et de traverser un hall. Trop court, le visiteur est prévenu alors qu'il ne * peut plus y être ; trop long, il oublie entre le rappel et la conférence. * Cela dépend de la taille du lieu, ce qui ne se devine pas depuis le plan — * d'où le réglage, à côté du temps de visite par stand. */ const RAPPEL_DEFAUT = 15; // minutes const RAPPEL_MIN = 2, RAPPEL_MAX = 120; const reglageRappel = () => conf("_rappel"); const rappelsVoulus = () => reglageRappel().actif === true; function minutesRappel(){ const v = Math.round(+reglageRappel().minutes); return v >= RAPPEL_MIN && v <= RAPPEL_MAX ? v : RAPPEL_DEFAUT; } /* ------------------------------------------------------------ Ce que l'appareil sait faire ------------------------------------------------------------ */ const RAPPELS_API = API ? API.replace(/[^/]*$/, "rappels") : ""; /** Le consentement se retient ici, à côté du parcours qu'il accompagne. * L'autorisation du navigateur ne suffit pas à le dire : elle a pu être * accordée pour ce salon-ci et l'interrupteur éteint depuis. */ const cleRappels = () => "plan-rappels:" + (SLUG || (DATA && DATA.evenement) || ""); let RAPPELS_ACTIFS = false; function chargeRappels(){ try { RAPPELS_ACTIFS = localStorage.getItem(cleRappels()) === "1"; } catch (e) { RAPPELS_ACTIFS = false; } } function retientRappels(){ try { if (RAPPELS_ACTIFS) localStorage.setItem(cleRappels(), "1"); else localStorage.removeItem(cleRappels()); } catch (e) {} } /** L'appareil peut-il recevoir un rappel ? Trois objets doivent exister, et * leur absence est la façon dont un iPhone non installé se signale. */ const poussePossible = () => !!( "serviceWorker" in navigator && "PushManager" in window && "Notification" in window); /** Le rappel a-t-il lieu d'être proposé ici ? La page à données figées n'a pas * d'API, l'administration n'a pas de visiteur, et l'exploitant peut ne pas * vouloir de la commande. */ const rappelsOfferts = () => !!( RAPPELS_API && SLUG && rappelsVoulus() && !BORNE && document.documentElement.dataset.role !== "admin"); /** Un plan ouvert dans un onglet iOS : le seul cas où l'absence de `PushManager` * se répare, et où dire comment vaut mieux que se taire. */ const iOSsansInstallation = () => !poussePossible() && /iP(hone|od|ad)|Macintosh/.test(navigator.userAgent) && "ontouchend" in document && supportMesure() === "web"; /* ------------------------------------------------------------ L'instant d'une conférence ------------------------------------------------------------ */ /** * L'heure absolue, alors que tout le reste de la page raisonne en heure locale * du salon. * * Il le faut : une minuterie se pose sur un instant, pas sur un cadran. * * On part de l'heure que la page affiche, et non de la forme ISO que les * données portent parfois à côté. C'est délibéré : le tiroir annonce « 14h30 », * et un rappel calculé sur une autre source tomberait à une autre minute le * jour où les deux se contrediraient — un programme ressaisi à la main, un * décalage mal posé à la source. Ce qui est promis est ce qui est écrit. * * Reste à savoir ce que le fuseau du salon valait *ce jour-là*, changement * d'heure compris. D'où l'aller-retour : on suppose l'heure murale exprimée en * UTC, on regarde ce que le fuseau en aurait affiché, et on corrige de l'écart. * Deux tours, parce que la correction peut elle-même franchir le changement * d'heure — un dimanche de mars à deux heures et demie du matin. * * Un fuseau que le navigateur ne connaît pas ne donne pas d'instant du tout, et * la conférence ne se rappelle pas : c'est la règle que `momentLocal` applique * déjà en refusant d'afficher une heure plutôt que d'en afficher une fausse. */ function instantAbsolu(c){ const p = instantConf(c); if (!p) return NaN; const mural = Date.UTC(p.an, p.mois - 1, p.jour, +p.h, +p.min); let t = mural; for (let i = 0; i < 2; i++){ const lu = momentLocal(new Date(t).toISOString(), DATA.fuseau); if (!lu) return NaN; t = mural - (Date.UTC(lu.an, lu.mois - 1, lu.jour, +lu.h, +lu.min) - t); } return t; } /** Les conférences retenues qui n'ont pas encore commencé. Le reste du tiroir * les montre toutes — on ne retire pas de sa liste ce qu'on vient de voir — * mais il n'y a rien à rappeler d'une conférence commencée. */ function confsARappeler(){ const maintenant = Date.now(); return PARCOURS.confs.map(id => CONFS.get(id)).filter(Boolean) .map(c => ({ c: c, t: instantAbsolu(c) })) .filter(r => !isNaN(r.t) && r.t > maintenant); } /** * Ce qu'on demande au serveur de garder, et rien de plus. * * Un rappel dont l'heure est déjà passée ne part pas : le visiteur a pu poser * son interrupteur dix minutes avant une conférence qu'il a de toute façon * sous les yeux. La durée de vie du message vaut le délai lui-même — passé ce * point, un téléphone rallumé annoncerait une conférence commencée, et mieux * vaut que le message soit jeté que délivré. */ function rappelsDuParcours(){ const n = minutesRappel(), avant = n * 60000, maintenant = Date.now(); return confsARappeler() .filter(r => r.t - avant > maintenant) .map(r => { const d = instantConf(r.c); const detail = [traduit("Dans " + n + " min"), d ? d.h + "h" + d.min : "", r.c.salle || ""].filter(Boolean).join(" · "); return { conf: String(r.c.id), envoi_a: new Date(r.t - avant).toISOString(), titre: r.c.nom || traduit("Conférence"), corps: detail, adresse: adresseDuRappel(r.c.id), vie: n * 60, }; }); } /** Où mène la notification : cette page, ce salon, cette conférence. L'adresse * est absolue — le service de second plan l'ouvrira sans contexte. */ function adresseDuRappel(id){ const u = new URL(location.href); u.hash = ""; u.searchParams.set("conf", id); return u.href; } /* ------------------------------------------------------------ S'abonner, et se tenir à jour ------------------------------------------------------------ */ /** La clé publique du serveur, sous la forme d'octets : Safari n'accepte pas * la chaîne base64url que Chrome tolère. */ function octetsDeCle(b64){ const net = String(b64).replace(/-/g, "+").replace(/_/g, "/"); const brut = atob(net + "=".repeat((4 - net.length % 4) % 4)); const o = new Uint8Array(brut.length); for (let i = 0; i < brut.length; i++) o[i] = brut.charCodeAt(i); return o; } async function abonnementCourant(){ const reg = await navigator.serviceWorker.getRegistration(); return reg ? await reg.pushManager.getSubscription() : null; } async function abonne(){ const reg = await navigator.serviceWorker.ready; const deja = await reg.pushManager.getSubscription(); if (deja) return deja; const r = await fetch(RAPPELS_API + "/cle"); if (!r.ok) throw new Error("clé indisponible"); const cle = (await r.json()).cle; if (!cle) throw new Error("clé absente"); /* `userVisibleOnly` n'est pas négociable : le navigateur n'accorde le Web Push qu'à qui s'engage à montrer chaque message reçu. Cela tombe bien — c'est exactement ce qu'on veut en faire. */ return await reg.pushManager.subscribe({ userVisibleOnly: true, applicationServerKey: octetsDeCle(cle) }); } let _syncRappels = null; /** * Renvoyer la liste entière plutôt que ce qui a changé. * * C'est la même idée que le tiroir, qui se relit en entier à chaque affichage : * une conférence retirée disparaît d'elle-même, et il n'y a pas de geste à ne * pas manquer. L'envoi est différé de quelques instants parce qu'il suit le * rafraîchissement du parcours, qui peut se produire trois fois de suite pour * un seul geste. */ function synchroniseRappels(){ if (!RAPPELS_ACTIFS || !rappelsOfferts() || !poussePossible()) return; clearTimeout(_syncRappels); _syncRappels = setTimeout(async () => { try { const ab = await abonnementCourant(); if (!ab) return; await fetch(RAPPELS_API, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ slug: SLUG, abonnement: ab.toJSON(), rappels: rappelsDuParcours() }), }); } catch (e) { /* Une panne réseau ne se dit pas : le visiteur n'y peut rien, et ce qu'il a demandé reste demandé — la prochaine ouverture du plan renverra la liste. Le pire cas est un rappel qui ne part pas, pas une visite gâchée. */ } }, 400); } /** Éteindre : l'abonnement est rendu au navigateur, et le serveur oublie ce * qu'il gardait. Dans cet ordre — un abonnement rendu avant l'effacement * laisserait des rangs qu'on ne saurait plus nommer. */ async function eteintRappels(){ const ab = await abonnementCourant().catch(() => null); if (ab){ try { await fetch(RAPPELS_API, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ slug: SLUG, abonnement: ab.toJSON(), rappels: [] }), }); } catch (e) {} try { await ab.unsubscribe(); } catch (e) {} } RAPPELS_ACTIFS = false; retientRappels(); } async function allumeRappels(){ /* La demande doit suivre le geste sans rien attendre entre les deux : un `await` placé avant ferait perdre au navigateur le lien entre l'appui et la question, et Safari refuse alors d'afficher quoi que ce soit. */ const reponse = await Notification.requestPermission(); if (reponse !== "granted") return false; await abonne(); RAPPELS_ACTIFS = true; retientRappels(); synchroniseRappels(); return true; } /* ------------------------------------------------------------ L'interrupteur, au pied des conférences retenues ------------------------------------------------------------ */ /** Une ligne d'explication, du même gris que le reste du tiroir. */ function aideRappel(hote, texte){ const p = document.createElement("p"); p.className = "rapAide"; p.textContent = texte; hote.appendChild(p); return p; } /** * Ce que le tiroir montre, selon ce que l'appareil permet. * * Quatre états, et un seul porte un interrupteur. Un interrupteur mort — grisé, * inerte — en dirait moins qu'une phrase : le visiteur d'un iPhone non installé * a quelque chose à faire, celui qui a refusé les notifications a un réglage à * rouvrir, et ni l'un ni l'autre ne se lit dans une case qui ne bouge pas. */ function poseRappels(hote){ if (!rappelsOfferts()) return; if (!confsARappeler().length) return; const bloc = document.createElement("div"); bloc.className = "rapBloc"; if (iOSsansInstallation()){ aideRappel(bloc, "Pour être prévenu avant vos conférences, ajoutez le plan " + "à votre écran d'accueil : sur iPhone, les notifications n'existent que là."); hote.appendChild(bloc); return; } if (!poussePossible()) return; // ni rappel ni remède : rien à dire if (Notification.permission === "denied"){ aideRappel(bloc, "Les notifications sont bloquées pour ce site. Vous pouvez " + "les réautoriser dans les réglages de votre navigateur."); hote.appendChild(bloc); return; } const n = minutesRappel(); const l = document.createElement("label"); l.className = "rapBascule"; l.innerHTML = ''; l.querySelector(".t").textContent = "Me prévenir " + n + " min avant"; const case_ = l.querySelector("input"); case_.checked = RAPPELS_ACTIFS; bloc.appendChild(l); const aide = aideRappel(bloc, RAPPELS_ACTIFS ? "Une notification par conférence retenue, même plan fermé." : "Une notification par conférence retenue, même plan fermé. Les heures " + "et les titres retenus partent alors sur nos serveurs, et s'effacent " + "après le salon."); case_.onchange = async () => { case_.disabled = true; try { if (case_.checked){ const ok = await allumeRappels(); if (!ok){ case_.checked = false; aide.textContent = Notification.permission === "denied" ? "Les notifications sont bloquées pour ce site. Vous pouvez les " + "réautoriser dans les réglages de votre navigateur." : "Les notifications n'ont pas été autorisées."; } else { aide.textContent = "Une notification par conférence retenue, même plan fermé."; } } else { await eteintRappels(); aide.textContent = "Une notification par conférence retenue, même plan fermé. " + "Les heures et les titres retenus partent alors sur nos serveurs, et " + "s'effacent après le salon."; } } catch (e) { case_.checked = false; aide.textContent = "Le rappel n'a pas pu être posé. Réessayez dans un instant."; } case_.disabled = false; }; hote.appendChild(bloc); } /** * Au démarrage : reprendre ce qui avait été demandé. * * L'abonnement peut avoir été renouvelé par le navigateur entre deux visites, * ou perdu — des données de site effacées, une icône retirée puis remise. On * refait donc le tour plutôt que de faire confiance à ce qu'on avait retenu, et * la liste repart avec, à jour des conférences ajoutées depuis. */ async function reprendRappels(){ chargeRappels(); if (!RAPPELS_ACTIFS || !rappelsOfferts() || !poussePossible()) return; if (Notification.permission !== "granted"){ // l'autorisation a été retirée depuis : l'interrupteur ne ment pas RAPPELS_ACTIFS = false; retientRappels(); return; } try { await abonne(); } catch (e) { return; } synchroniseRappels(); }
