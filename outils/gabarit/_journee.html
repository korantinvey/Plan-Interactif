/* ============================================================ 11 ter. Organiser sa journée ============================================================ */ /** * Le parcours dit ce qu'on veut voir. Il ne dit ni dans quel ordre, ni si la * journée y suffit — et c'est pourtant là que tout se joue : douze stands * retenus au hasard des allées, deux conférences à heure fixe, et un visiteur * qui repart en ayant traversé quatre fois le même hall sans voir la moitié * de sa liste. * * On lui demande donc une seule chose — à quelle heure il arrive — et on * range le reste : les conférences sont des rendez-vous qu'on ne déplace pas, * les stands se glissent entre elles, et l'ordre retenu est celui qui fait le * moins de chemin. Le temps passé sur un stand ne se devine pas depuis le * plan ; c'est l'exploitant qui le fixe, salon par salon. * * Ce qu'on rend n'est pas une promesse : une conversation s'éternise, une * conférence commence en retard, et l'heure de seize heures ne vaudra plus * rien à quatorze. C'est un ordre de visite, avec des heures qui disent ce * qui tient et ce qui ne tient pas. */ /** * Ce que coûte un changement de pavillon. * * Les données ne disent rien de ce qui relie deux halls — ni passerelle, ni * allée extérieure, ni distance — et l'itinéraire s'y coupe franchement * plutôt que d'inventer un tracé. Ordonner une journée demande pourtant un * prix : sans lui, l'algorithme ferait l'aller-retour entre deux pavillons * comme il change d'allée, et rendrait un programme absurde. * * Ce forfait n'est donc pas une mesure, c'est une dissuasion : plus cher que * n'importe quel détour à l'intérieur d'un seul hall, ce qui suffit à * regrouper les visites pavillon par pavillon. Il ne compte jamais dans la * distance annoncée — celle-ci ne porte que ce qui est tracé, et le dit. */ const TRANSFERT_M = 300; // pour ordonner, jamais pour annoncer const TRANSFERT_MIN = 5; // ce qu'on réserve dans l'horaire /* L'heure d'arrivée d'une visite à l'autre : celui qui vient deux jours de suite vient rarement à deux heures différentes. */ const CLE_ARRIVEE = "plan-journee-arrivee"; let JOURNEE = null; // la dernière journée calculée let vueJournee = false; // le tiroir montre-t-il la journée ou la liste ? /* ------------------------------------------------------------ Les heures ------------------------------------------------------------ */ const minutesDe = (d) => d ? (+d.h) * 60 + (+d.min) : null; const finInstant = (c) => momentLocal(c.finLocal, DATA.fuseau) || momentLocal(c.fin, DATA.fuseau); function ecritHeure(m){ // une journée qui déborde sur la nuit repasse par zéro plutôt que d'écrire 26h const t = ((Math.round(m) % 1440) + 1440) % 1440; return String(Math.floor(t / 60)).padStart(2, "0") + "h" + String(t % 60).padStart(2, "0"); } function ecritMinutes(m){ const t = Math.max(1, Math.round(m)); if (t < 60) return t + " min"; const h = Math.floor(t / 60), r = t % 60; return h + " h" + (r ? " " + r + " min" : ""); } /** Les jours où le salon a un programme, dans l'ordre. */ function joursSalon(){ const vus = new Map(); CONFS.forEach(c => { const d = instantConf(c); if (d && !vus.has(d.cle)) vus.set(d.cle, { cle: d.cle, nom: jourCourt(d) }); }); return [...vus.values()].sort((a, b) => a.cle.localeCompare(b.cle)); } /** Le jour sur lequel la fenêtre s'ouvre : celui de la première conférence * retenue, à défaut le premier du salon. */ function jourPropose(){ const d = PARCOURS.confs.map(id => CONFS.get(id)).filter(Boolean) .map(instantConf).filter(Boolean) .sort((a, b) => a.cle.localeCompare(b.cle))[0]; if (d) return d.cle; const j = joursSalon(); return j.length ? j[0].cle : ""; } /* ------------------------------------------------------------ Les points de la journée ------------------------------------------------------------ */ /** Là où se tient une conférence : sa zone, qui est un objet du plan. Une * conférence qui n'en a pas garde son heure et sa place dans la journée — * on ne sait simplement pas y mener. */ function pointConf(c){ const z = c.zone && parId.get(String(c.zone)); return z ? pointObjet(z) : null; } /** Les départs qu'on propose : les repères du salon, les entrées en tête — * c'est par là qu'on arrive, et c'est le seul endroit dont un visiteur sans * géolocalisation est sûr. */ function departsProposes(){ const l = []; DATA.plans.forEach(p => reperesDe(p).forEach(f => { const pic = f.picto || pictoDe(f.txt); l.push({ pt: pointRepere(f, p), entree: pic === "entree" ? 1 : 0 }); })); l.sort((a, b) => b.entree - a.entree); return l.slice(0, 20); } /** * De chaque arrêt à tous les autres. * * Un balayage en largeur par arrêt, et non un A* par paire : vingt arrêts * font quatre cents paires, et le hall entier coûte moins cher à balayer * qu'un seul chemin à chercher. La longueur obtenue est celle des allées, * sans prix du virage ni du bord — de quoi comparer deux ordres de visite, * ce qu'on lui demande. Le tracé montré au visiteur reste celui de `troncon`. */ function matriceJournee(pts, pmr){ const n = pts.length; const D = []; for (let i = 0; i < n; i++) D.push(new Float64Array(n)); /* Un arrêt qu'aucune allée n'atteint — un stand enclavé, un repère posé dans un mur — ne doit pas sortir de la journée pour autant : on retombe sur la distance à vol d'oiseau, majorée du détour qu'une allée impose. */ const volDoiseau = (a, b) => Math.hypot(a.xy[0] - b.xy[0], a.xy[1] - b.xy[1]) * 1.3; const parPlan = new Map(); pts.forEach((pt, i) => { if (!pt) return; if (!parPlan.has(pt.p)) parPlan.set(pt.p, []); parPlan.get(pt.p).push(i); }); /* Ce qui sépare un arrêt de la porte la plus proche de son pavillon : c'est par là que passe tout trajet vers un autre hall. */ const auxPortes = new Float64Array(n); parPlan.forEach((idx, ip) => { const p = DATA.plans[ip]; const g = grille(p, pmr); const cases = idx.map(i => accroche(g, pts[i].xy)); const portes = portesDe(p).map(f => accroche(g, f.pts[0])) .filter(c => c >= 0 && g.nappe[c]); idx.forEach((i, k) => { const d = distancesDepuis(g, cases[k]); idx.forEach((j, l) => { if (j === i) return; const c = cases[l]; D[i][j] = (c >= 0 && d[c] >= 0) ? d[c] * PAS_GRILLE : volDoiseau(pts[i], pts[j]); }); let mieux = -1; portes.forEach(c => { if (d[c] >= 0 && (mieux < 0 || d[c] < mieux)) mieux = d[c]; }); auxPortes[i] = mieux >= 0 ? mieux * PAS_GRILLE : 0; }); }); for (let i = 0; i < n; i++) for (let j = 0; j < n; j++){ if (i === j || !pts[i] || !pts[j] || pts[i].p === pts[j].p) continue; D[i][j] = auxPortes[i] + TRANSFERT_M + auxPortes[j]; } return D; } /* ------------------------------------------------------------ L'ordre de visite ------------------------------------------------------------ */ /** * Ranger les stands entre les conférences. * * Le problème est celui du voyageur de commerce, avec des rendez-vous en * plus : l'ordre exact demanderait d'essayer toutes les permutations, et * vingt stands en font deux milliards de milliards. On construit donc au plus * juste — à chaque tour, le stand dont l'insertion coûte le moins de mètres, * là où elle tient dans le temps disponible — puis on reprend l'ensemble par * deux mouvements qui rattrapent l'essentiel : retourner un morceau de * créneau, et déplacer un arrêt ailleurs. * * Un stand qui ne tient nulle part reste dehors et se dit : mieux vaut une * liste courte et vraie qu'un programme intenable. */ function rangeJournee(creneaux, restants, dist, tempsCreneau){ const distanceCreneau = (c) => { let m = 0, prec = c.de; for (const s of c.stands){ m += dist(prec, s); prec = s; } return m + dist(prec, c.vers); }; const tient = (c) => c.tMax === null || c.t0 + tempsCreneau(c, distanceCreneau(c)) <= c.tMax; while (restants.length){ let choix = null; for (const s of restants){ for (const c of creneaux){ for (let k = 0; k <= c.stands.length; k++){ const avant = k === 0 ? c.de : c.stands[k - 1]; const apres = k === c.stands.length ? c.vers : c.stands[k]; const cout = dist(avant, s) + dist(s, apres) - dist(avant, apres); if (choix && cout >= choix.cout) continue; c.stands.splice(k, 0, s); const ok = tient(c); c.stands.splice(k, 1); if (ok) choix = { cout: cout, c: c, k: k, s: s }; } } } if (!choix) break; choix.c.stands.splice(choix.k, 0, choix.s); restants.splice(restants.indexOf(choix.s), 1); } /* Les deux reprises. Chaque mouvement se juge sur ses seules arêtes — la longueur entière du créneau n'a pas à être refaite — et n'est retenu que s'il fait gagner du chemin sans faire perdre un rendez-vous. */ const bord = (c, k) => k === 0 ? c.de : c.stands[k - 1]; const suite = (c, k) => k >= c.stands.length ? c.vers : c.stands[k]; for (let tour = 0; tour < 30; tour++){ let gagne = false; for (const c of creneaux){ for (let i = 0; i < c.stands.length; i++){ for (let j = i + 1; j < c.stands.length; j++){ const a = bord(c, i), b = suite(c, j + 1); const gain = dist(a, c.stands[i]) + dist(c.stands[j], b) - dist(a, c.stands[j]) - dist(c.stands[i], b); if (gain <= 1e-9) continue; const copie = c.stands.slice(); const morceau = c.stands.slice(i, j + 1).reverse(); c.stands.splice(i, j - i + 1, ...morceau); if (tient(c)) gagne = true; else c.stands = copie; } } } for (const c of creneaux){ for (let i = 0; i < c.stands.length; i++){ const s = c.stands[i]; const retrait = dist(bord(c, i), s) + dist(s, suite(c, i + 1)) - dist(bord(c, i), suite(c, i + 1)); c.stands.splice(i, 1); let mieux = null; for (const d of creneaux){ for (let k = 0; k <= d.stands.length; k++){ const pose = dist(bord(d, k), s) + dist(s, suite(d, k)) - dist(bord(d, k), suite(d, k)); if (pose >= retrait - 1e-9 || (mieux && pose >= mieux.pose)) continue; d.stands.splice(k, 0, s); const ok = tient(d); d.stands.splice(k, 1); if (ok) mieux = { pose: pose, d: d, k: k }; } } if (mieux){ mieux.d.stands.splice(mieux.k, 0, s); gagne = true; } else c.stands.splice(i, 0, s); } } if (!gagne) break; } } /* ------------------------------------------------------------ La journée, calculée ------------------------------------------------------------ */ /** * Ce que le visiteur a demandé : une heure d'arrivée, un jour, un point de * départ. Ce qu'on lui rend : un déroulé, un tracé, et la liste franche de ce * qui n'a pas trouvé sa place. */ function calculeJournee(choix){ const pmr = ITI.pmr; const tv = minutesVisite(); const allure = pmr ? ALLURE_PMR : ALLURE; const enMinutes = (m) => m / (allure * 60); const stands = PARCOURS.stands.map(id => parId.get(id)).filter(Boolean); const horsJour = [], passees = [], sansLieu = []; const retenues = PARCOURS.confs.map(id => CONFS.get(id)).filter(Boolean) .map(c => ({ c: c, d: instantConf(c), f: finInstant(c) })) .sort((a, b) => (minutesDe(a.d) || 0) - (minutesDe(b.d) || 0)); const rdv = []; retenues.forEach(x => { if (!x.d || (choix.jour && x.d.cle !== choix.jour)){ horsJour.push(x.c); return; } const t0 = minutesDe(x.d); /* Une conférence commencée avant l'arrivée n'est plus un rendez-vous : on ne la fait pas manquer, on la dit manquée. */ if (t0 < choix.arrivee){ passees.push(x.c); return; } const pt = pointConf(x.c); if (!pt) sansLieu.push(x.c); /* Sans heure de fin, on compte une heure : enchaîner aussitôt ferait un programme faux, et attendre la journée entière un programme vide. */ const t1 = x.f ? Math.max(t0 + 5, minutesDe(x.f)) : t0 + 60; rdv.push({ c: x.c, pt: pt, t0: t0, t1: t1 }); }); /* Les points, dans un seul tableau : le départ, les stands, les salles. C'est leur rang qui sert d'identité partout ensuite. */ const PTS = []; const iDepart = choix.depart ? PTS.push(choix.depart) - 1 : -1; const iStands = stands.map(o => PTS.push(pointObjet(o)) - 1); rdv.forEach(r => { r.i = PTS.push(r.pt) - 1; }); const D = matriceJournee(PTS, pmr); /* Un bout de trajet dont on ignore un des deux bouts ne coûte rien : le départ « peu importe » et une salle non située ne sont pas des détours, ce sont des trous. */ const dist = (a, b) => (a < 0 || b < 0 || !PTS[a] || !PTS[b]) ? 0 : D[a][b]; const tempsCreneau = (c, m) => enMinutes(m) + c.stands.length * tv; /* Les intervalles entre rendez-vous, c'est là qu'on range les stands. Une conférence dont la salle n'est située nulle part borne l'heure sans borner le lieu : on ne sait pas s'en approcher, et on repart d'où l'on était. */ const creneaux = []; let de = iDepart, t0 = choix.arrivee; rdv.forEach(r => { creneaux.push({ de: de, t0: t0, vers: r.pt ? r.i : -1, tMax: r.t0, stands: [] }); if (r.pt) de = r.i; t0 = r.t1; }); creneaux.push({ de: de, t0: t0, vers: -1, tMax: null, stands: [] }); const restants = iStands.slice(); rangeJournee(creneaux, restants, dist, tempsCreneau); /* L'ordre est arrêté : on le déroule en heures et en tronçons. Les mètres annoncés sont ceux du tracé, pas ceux de la matrice — c'est le trait que le visiteur suivra. */ const etapes = [], arrets = [], trace = [], retards = []; let t = choix.arrivee, m = 0, coupe = false, rang = 0; const rejoint = (a, b) => { if (a < 0 || b < 0 || !PTS[a] || !PTS[b]) return; const pa = DATA.plans[PTS[a].p], pb = DATA.plans[PTS[b].p]; if (pa === pb){ const tr = troncon(pa, pmr, PTS[a].xy, PTS[b].xy); const l = tr ? tr.m : dist(a, b); if (tr) trace.push(tr); etapes.push({ genre: "marche", m: l, min: enMinutes(l), trace: !!tr }); t += enMinutes(l); m += l; return; } coupe = true; const sortie = versPorte(pa, pmr, PTS[a].xy, false); const entree = versPorte(pb, pmr, PTS[b].xy, true); if (sortie){ trace.push(sortie); m += sortie.m; } trace.push({ genre: "transfert", de: pa.libelle, vers: pb.libelle }); if (entree){ trace.push(entree); m += entree.m; } const min = enMinutes((sortie ? sortie.m : 0) + (entree ? entree.m : 0)) + TRANSFERT_MIN; etapes.push({ genre: "transfert", de: pa.libelle, vers: pb.libelle, min: min }); t += min; }; const pose = (i, conf) => { rang++; if (PTS[i]) arrets.push({ p: PTS[i].p, xy: PTS[i].xy, n: String(rang), conf: !!conf }); return rang; }; if (iDepart >= 0){ arrets.push({ p: PTS[iDepart].p, xy: PTS[iDepart].xy, n: "", depart: true }); etapes.push({ genre: "depart", nom: choix.depart.nom, detail: choix.depart.detail, t: t }); } let ou = iDepart; creneaux.forEach((c, k) => { c.stands.forEach(s => { rejoint(ou, s); const n = pose(s, false); etapes.push({ genre: "stand", n: n, o: parId.get(PTS[s].id), pt: PTS[s], t0: t, t1: t + tv }); t += tv; ou = s; }); const r = rdv[k]; if (!r) return; rejoint(ou, r.i); if (t > r.t0 + 1) retards.push({ c: r.c, min: t - r.t0 }); else if (t < r.t0) etapes.push({ genre: "attente", min: r.t0 - t }); t = Math.max(t, r.t0); const n = pose(r.i, true); etapes.push({ genre: "conf", n: n, c: r.c, pt: r.pt, t0: r.t0, t1: r.t1 }); t = r.t1; /* Une conférence dont la salle n'est rattachée à aucune zone ne dit pas où l'on se trouve après elle : on repart du dernier endroit connu, sans quoi le trajet suivant — un changement de pavillon, souvent — disparaîtrait du déroulé sans un mot. */ if (r.pt) ou = r.i; }); return { jour: choix.jour, jourNom: choix.jourNom, arrivee: choix.arrivee, fin: t, pmr: pmr, tv: tv, etapes: etapes, arrets: arrets, trace: trace, m: m, coupe: coupe, nArrets: rang, restants: restants.map(i => parId.get(PTS[i].id)).filter(Boolean), horsJour: horsJour, passees: passees, sansLieu: sansLieu, retards: retards, }; } /* ------------------------------------------------------------ Ce que la journée montre ------------------------------------------------------------ */ function rangJournee(hote, heure, dedans, classe){ const d = document.createElement("div"); d.className = "jRang" + (classe ? " " + classe : ""); const h = document.createElement("span"); h.className = "jH"; h.textContent = heure || ""; d.appendChild(h); d.appendChild(dedans); hote.appendChild(d); return d; } function lienJournee(hote, texte){ const s = document.createElement("span"); s.className = "jL"; s.textContent = texte; rangJournee(hote, "", s, "lien"); } function arretJournee(hote, e){ const b = document.createElement("button"); b.type = "button"; b.className = "jArret" + (e.genre === "conf" ? " conf" : ""); b.innerHTML = '' + ''; b.querySelector(".jN").textContent = e.n; b.querySelector(".jD").textContent = ecritMinutes(e.t1 - e.t0); if (e.genre === "conf"){ if (e.c.couleur) b.style.setProperty("--tc", e.c.couleur); b.querySelector(".jT").textContent = e.c.nom; b.querySelector(".jS").textContent = [e.c.salle, e.pt ? e.pt.detail : ""] .filter(Boolean).join(" · ") || "Salle non située sur le plan"; b.onclick = () => ficheConf(e.c.id, "parcours"); } else { b.querySelector(".jT").textContent = e.pt.nom; b.querySelector(".jS").textContent = e.pt.detail; b.onclick = () => { if (e.o) select(e.o.id, true, "parcours"); }; } rangJournee(hote, ecritHeure(e.t0), b); } function remplitJournee(){ const hote = $("jCorps"); hote.innerHTML = ""; const j = JOURNEE; if (!j) return; /* Les notes et les alertes sont celles de l'itinéraire, classes comprises : c'est la même voix qui parle du même calcul, à deux endroits du tiroir. */ const dit = (cls, txt) => { const p = document.createElement("p"); p.className = cls; p.textContent = txt; hote.appendChild(p); return p; }; if (j.perime) dit("iAlerte", "Votre parcours a changé depuis ce calcul : " + "reprenez-le pour que la journée en tienne compte."); const bilan = document.createElement("div"); bilan.className = "jBilan"; bilan.innerHTML = ''; bilan.querySelector("b").textContent = ecritHeure(j.arrivee) + " – " + ecritHeure(j.fin); bilan.querySelector("span").textContent = [j.nArrets + (j.nArrets > 1 ? " arrêts" : " arrêt"), (j.coupe ? "au moins " : "") + ecritDistance(j.m) + " de marche", j.pmr ? "itinéraire accessible" : ""].filter(Boolean).join(" · "); hote.appendChild(bilan); const fil = document.createElement("div"); fil.className = "jFil"; hote.appendChild(fil); j.etapes.forEach(e => { if (e.genre === "depart"){ const s = document.createElement("span"); s.className = "jL"; s.textContent = "Départ : " + e.nom; rangJournee(fil, ecritHeure(e.t), s, "lien"); } else if (e.genre === "marche"){ lienJournee(fil, ecritDistance(e.m) + " · " + ecritMinutes(e.min) + " de marche" + (e.trace ? "" : " (non tracé)")); } else if (e.genre === "transfert"){ lienJournee(fil, "Rejoignez le " + e.vers + " · comptez " + ecritMinutes(e.min) + ", ce trajet n'est pas sur le plan"); } else if (e.genre === "attente"){ lienJournee(fil, ecritMinutes(e.min) + " sur place avant le début"); } else { arretJournee(fil, e); } }); /* Ce qui n'entre pas dans la journée se dit, et se dit précisément : un rang laissé de côté sans un mot passerait pour un oubli du calcul. Ces listes empruntent leurs rangs au parcours — même vignette, même croix : ce sont les mêmes choses, vues d'ailleurs. */ const groupe = (titre, liste, rendu) => { if (!liste.length) return; const g = document.createElement("div"); g.className = "pGroupe jReste"; g.innerHTML = ''; g.querySelector(".eyebrow").textContent = titre; liste.forEach(x => rangParcours(g, rendu(x))); hote.appendChild(g); }; /* Un rendez-vous qu'on n'atteint pas ne se raye pas tout seul : le visiteur décidera ce qu'il sacrifie, pas nous. */ j.retards.forEach(r => dit("iAlerte", "« " + r.c.nom + " » : au mieux, vous y " + "arriveriez " + ecritMinutes(r.min) + " après le début — le trajet n'y suffit " + "pas. Arrivez plus tôt, ou retirez ce qui la précède.")); /* En l'état cette liste reste vide : la journée n'a pas d'heure de fin, si bien qu'un stand refusé entre deux conférences finit toujours par tenir après la dernière. C'est le garde-fou de la contrainte horaire — le jour où l'on demandera aussi à quelle heure le visiteur repart, c'est ici qu'il lira ce qui n'y tient pas, plutôt que de recompter ses stands. */ groupe("Restés de côté", j.restants, o => ({ genre: "stand", id: o.id, dessus: false, titre: o.nom || o.plan || ("Stand " + (o.code || "")).trim(), detail: DATA.plans[o.p].libelle + (o.code ? " · " + o.code : ""), ouvre: () => select(o.id, true, "parcours"), })); groupe("Un autre jour", j.horsJour, c => ({ genre: "conf", id: c.id, couleur: c.couleur, dessus: true, titre: c.nom, detail: jourCourt(instantConf(c)) || "date inconnue", ouvre: () => ficheConf(c.id, "parcours"), })); groupe("Déjà commencées à votre arrivée", j.passees, c => ({ genre: "conf", id: c.id, couleur: c.couleur, dessus: true, titre: c.nom, detail: (instantConf(c) ? instantConf(c).h + "h" + instantConf(c).min : "") + (c.salle ? " · " + c.salle : ""), ouvre: () => ficheConf(c.id, "parcours"), })); if (j.sansLieu.length) dit("iNote", j.sansLieu.length > 1 ? "Certaines conférences retenues ne sont rattachées à aucune zone du plan : " + "elles gardent leur heure, mais le trajet ne peut pas y mener." : "« " + j.sansLieu[0].nom + " » n'est rattachée à aucune zone du plan : elle " + "garde son heure, mais le trajet ne peut pas y mener."); dit("iNote", "Comptez " + ecritMinutes(j.tv) + " par stand. Les heures sont " + "indicatives : elles supposent qu'on ne s'attarde pas, et ne tiennent compte " + "ni des files ni des pauses."); } /* ------------------------------------------------------------ Les deux vues du tiroir ------------------------------------------------------------ */ /** Le tiroir montre la liste ou la journée, jamais les deux : c'est la même * journée, lue avant puis après le calcul. */ function appliqueVueParcours(){ const sur = vueJournee && !!JOURNEE; $("pCorps").hidden = sur; $("pPied").hidden = sur; $("jCorps").hidden = !sur; $("jPied").hidden = !sur; $("pEyebrow").textContent = sur ?
