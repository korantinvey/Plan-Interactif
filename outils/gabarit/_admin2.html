Les en-têtes sont lus à l'appel, pas au démarrage : en * administration la session n'existe qu'une fois l'identité vérifiée, et * c'est elle qui donne accès aux événements encore en brouillon. */ function charge(){ annonce("Chargement du plan…"); const entetes = {}; if (document.documentElement.dataset.role === "admin"){ try { const cfg = JSON.parse(localStorage.getItem("console-config") || "null") || window.PLAN_CONFIG || null; const ses = JSON.parse(localStorage.getItem("console-session") || "null"); if (cfg?.anonKey && ses?.access_token){ entetes["apikey"] = cfg.anonKey; entetes["Authorization"] = "Bearer " + ses.access_token; } } catch (e) {} } return fetch(API + "?slug=" + encodeURIComponent(SLUG), { headers: entetes, cache: "no-store" }) .then(r => { // session périmée : on repasse par l'écran de connexion plutôt que // d'afficher une erreur dans laquelle l'exploitant ne peut rien faire if (r.status === 401 && typeof ecranAcces === "function"){ try { localStorage.removeItem("console-session"); } catch (e) {} ecranAcces("Session expirée, reconnectez-vous."); return Promise.reject(new Error("Session expirée.")); } return r.ok ?
