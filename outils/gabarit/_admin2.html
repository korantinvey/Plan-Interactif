Une panne du réseau doit se voir, pas laisser un écran vide. ------------------------------------------------------------------ */ function demarre(d){ indexe(d); montePlan(); fit(); liste(); // le plan est manipulable : le fond peut arriver derrière chargeFond(state.plan); const v = decodeURIComponent((location.hash || "").slice(1)).toLowerCase(); if (v && v !== "admin"){ const cible = TOUS.find(o => String(o.code || "").toLowerCase() === v); if (cible) setTimeout(() => select(cible.id, true, "lien"), 80); } } function annonce(txt, erreur){ const l = $("list"); l.innerHTML = '
'; l.querySelector(".empty").textContent = txt; if (erreur) l.querySelector(".empty").style.color = "var(--rouge, #C2372B)"; $("countTxt").textContent = erreur ? "Indisponible" : "Chargement…"; // plus rien n'arrivera : le titre neutre est définitif, il cesse d'attendre if (erreur) $("titre").classList.remove("attente"); } /** * Charge le plan. Les en-têtes sont lus à l'appel, pas au démarrage : en * administration la session n'existe qu'une fois l'identité vérifiée, et * c'est elle qui donne accès aux événements encore en brouillon. */ /** * En-têtes de l'appel. En administration on présente sa session : c'est elle * qui donne accès aux événements encore en brouillon. Ils sont lus à chaque * appel, pas au démarrage : la session n'existe qu'une fois l'identité vérifiée. */ function entetesApi(){ const e = {}; if (document.documentElement.dataset.role !== "admin") return e; try { const cfg = JSON.parse(localStorage.getItem("console-config") || "null") || window.PLAN_CONFIG || null; const ses = JSON.parse(localStorage.getItem("console-session") || "null"); if (cfg?.anonKey && ses?.access_token){ e["apikey"] = cfg.anonKey; e["Authorization"] = "Bearer " + ses.access_token; } } catch (err) {} return e; } /** * Le fond d'un pavillon : cinquante fois le poids des stands, et purement * décoratif. On le charge après coup, une seule fois par pavillon, et on * l'injecte s'il concerne encore celui qu'on regarde. Son adresse porte * l'horodatage de la dernière synchronisation : le navigateur le garde * indéfiniment et ne le redemande qu'après une mise à jour du plan. */ const _fonds = new Map(); function chargeFond(i){ const p = DATA?.plans?.[i]; if (!p || !API) return Promise.resolve(); if (p.fond.length === 0 || p.fond.every(c => c.svg !== undefined)) return Promise.resolve(); if (_fonds.has(p.id)) return _fonds.get(p.id); const q = API + "?slug=" + encodeURIComponent(SLUG) + "&fond=" + encodeURIComponent(p.id) + "&v=" + encodeURIComponent(DATA.genereLe || "0"); const promesse = fetch(q, { headers: entetesApi() }) .then(r => r.ok ? r.json() : Promise.reject(new Error("HTTP " + r.status))) .then(d => { const par = new Map((d.calques || []).map(c => [c.cle, c.svg])); p.fond.forEach(c => { c.svg = par.get(c.cle) || ""; }); // le pavillon a pu changer pendant le chargement if (state.plan === i){ monteHabillage(); ordonneDom(); appliqueApparence(); } }) .catch(e => { _fonds.delete(p.id); // une panne réseau ne doit pas être définitive console.warn("Fond de plan indisponible :", e.message); }); _fonds.set(p.id, promesse); return promesse; } function charge(){ annonce("Chargement du plan…"); const entetes = entetesApi(); // La demande a pu partir dès l'en-tête de la page, avant même que le reste // soit lu : on la récupère plutôt que d'en lancer une seconde. En // administration elle n'existe pas — il faut y présenter sa session. const depart = (!entetes.Authorization && window.__plan) || fetch( API + "?slug=" + encodeURIComponent(SLUG), // un brouillon ne doit pas rester dans le cache du navigateur { headers: entetes, cache: entetes.Authorization ? "no-store" : "default" }); window.__plan = null; return depart .then(r => { // session périmée : on repasse par l'écran de connexion plutôt que // d'afficher une erreur dans laquelle l'exploitant ne peut rien faire if (r.status === 401 && typeof ecranAcces === "function"){ try { localStorage.removeItem("console-session"); } catch (e) {} ecranAcces("Session expirée, reconnectez-vous."); return Promise.reject(new Error("Session expirée.")); } return r.ok ? r.json() : r.json().then(j => Promise.reject(new Error(j.erreur || ("HTTP " + r.status)))); }) .then(d => { if (!d.plans?.length) throw new Error("Aucun pavillon : lancez une synchronisation depuis la console."); demarre(d); }) .catch(e => annonce("Plan indisponible : " + e.message, true)); } const fige = document.getElementById("data").textContent.trim(); if (fige && fige.indexOf("__DATA__") < 0) demarre(JSON.parse(fige)); else if (!API) annonce("Aucune source de données configurée.", true); // en administration, on attend la session : sans elle les brouillons sont invisibles else if (document.documentElement.dataset.role === "admin") annonce("Vérification de l'identité…"); else charge(); // les mesures de texte dépendent des polices : on retrace une fois chargées // les polices peuvent arriver avant les données : on ne retrace que si // quelque chose est déjà affiché if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => { if (DATA && view){ _lg.clear(); libelles(); } }); addEventListener("resize", () => { if (DATA && view) fit(); }); /*__PORTE_ADMIN__*/
