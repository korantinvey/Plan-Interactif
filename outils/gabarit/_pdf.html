/* ============================================================ 11 ter. Le parcours emporté : un PDF ============================================================ */ /** * Le parcours vit dans le navigateur de celui qui l'a fait, et nulle part * ailleurs. C'est ce qui le rend simple — aucun compte à créer — et fragile : * il ne se transmet pas, ne s'imprime pas, et disparaît avec l'historique du * téléphone. Un fichier règle les trois d'un coup. Il s'envoie à un collègue, * se glisse dans une poche, et s'ouvre dans un hall où le réseau est saturé. * * Le PDF s'écrit ici, à la main. Aucune bibliothèque n'est chargée : la page * de démonstration est publiée telle quelle, sans rien pouvoir appeler * d'extérieur, et les autres sont servies en fichiers figés. Ce n'est pas un * renoncement — un PDF 1.4 n'est qu'une suite d'objets numérotés suivie de la * table de leurs positions, et les quatorze polices de base, Helvetica * comprise, n'ont pas à être embarquées. */ /* La feuille s'imprime : elle garde l'encre sombre sur papier blanc, quel que soit le thème de l'écran. Ce sont les teintes du plan en thème clair. */ const PDF_ENCRE = [.08, .09, .10]; // --ink const PDF_GRIS = [.49, .51, .52]; // --ink-3 const PDF_ACCENT = [.18, .29, .82]; // --accent const PDF_TRAIT = [.82, .84, .80]; // --line const PDF_DOUX = [.86, .88, .98]; // --accent-soft const PDF_PAPIER = [1, 1, 1]; const PDF_FOND = [.95, .95, .93]; // les zones, en fond de plan /* Arial fut dessinée aux mesures d'Helvetica, et les substituts installés sur les systèmes libres le sont aussi : mesurer dans le navigateur revient à mesurer dans la police du PDF, à un cheveu près. Cela évite d'embarquer la table des largeurs des deux fontes. */ const PDF_POLICE = ["400", "Helvetica, Arial, sans-serif"]; const PDF_POLICE_G = ["700", "Helvetica, Arial, sans-serif"]; /* ------------------------------------------------------------ Le rédacteur de PDF ------------------------------------------------------------ */ function nouveauPdf(){ const pages = []; let p = null; // la page où l'on écrit const n2 = v => String(Math.round(v * 100) / 100); const rvb = c => n2(c[0]) + " " + n2(c[1]) + " " + n2(c[2]); const ec = s => { p.flux += s + "\n"; }; /* Les polices de base s'écrivent en WinAnsi — le Latin-1 de Windows. Il couvre le français ; l'apostrophe courbe, les guillemets typographiques et les tirets cadratins vivent dans la plage 128-159, que ce tableau rattrape. Le reste devient un point d'interrogation, qui se voit, plutôt qu'un caractère faux, qui ne se voit pas. */ const WINANSI = { "€": 128, "‚": 130, "ƒ": 131, "„": 132, "…": 133, "†": 134, "‡": 135, "ˆ": 136, "‰": 137, "Š": 138, "‹": 139, "Œ": 140, "Ž": 142, "‘": 145, "’": 146, "“": 147, "”": 148, "•": 149, "–": 150, "—": 151, "˜": 152, "™": 153, "š": 154, "›": 155, "œ": 156, "ž": 158, "Ÿ": 159 }; /* Le dictionnaire d'information ne se lit pas en WinAnsi mais en PDFDocEncoding, où les mêmes octets nomment d'autres signes : « — » y devient « Š ». On y écrit donc en UTF-16, forme que la norme prévoit et que tout lecteur comprend — en hexadécimal, qui n'a rien à échapper. */ function chaineUtf16(t){ let s = ""; } function chaine(t){ let s = "("; for (const c of String(t)){ const u = c.codePointAt(0); let b = u < 256 ? u : (WINANSI[c] || 63); if (b < 32) b = 32; // la parenthèse fermante d'un nom d'exposant terminerait la chaîne if (b === 40 || b === 41 || b === 92) s += "\\"; s += String.fromCharCode(b); } return s + ")"; } function mesure(t, taille, gras, ecart){ const s = String(t == null ? "" : t); return largeur(s, gras ? PDF_POLICE_G : PDF_POLICE) * taille + (ecart || 0) * Math.max(0, s.length - 1); } function texte(t, x, y, o){ o = o || {}; const s = String(t == null ? "" : t).replace(/\s+/g, " ").trim(); if (!s) return; const taille = o.taille || 10, gras = !!o.gras, ecart = o.ecart || 0; const l = mesure(s, taille, gras, ecart); const ax = o.align === "centre" ? x - l / 2 : o.align === "droite" ? x - l : x; ec("BT " + rvb(o.couleur || PDF_ENCRE) + " rg /" + (gras ? "F2" : "F1") + " " + n2(taille) + " Tf " + n2(ecart) + " Tc 1 0 0 1 " + n2(ax) + " " + n2(y) + " Tm " + chaine(s) + " Tj ET"); } /** Le texte réparti sur la largeur donnée, et rien qui déborde. */ function coupe(t, large, taille, gras){ const mots = String(t == null ? "" : t).replace(/\s+/g, " ").trim().split(" "); const out = []; let l = ""; for (let m of mots){ /* Une adresse web, un nom accolé : plus large que la colonne, aucune espace où couper. On tranche au caractère plutôt que de déborder. */ while (mesure(m, taille, gras) > large && m.length > 1){ let i = 1; while (i < m.length && mesure(m.slice(0, i + 1), taille, gras) <= large) i++; if (l){ out.push(l); l = ""; } out.push(m.slice(0, i)); m = m.slice(i); } const essai = l ? l + " " + m : m; if (!l || mesure(essai, taille, gras) <= large) l = essai; else { out.push(l); l = m; } } if (l) out.push(l); return out.length ? out : [""]; } const rect = (x, y, w, h, couleur) => ec("q " + rvb(couleur) + " rg " + n2(x) + " " + n2(y) + " " + n2(w) + " " + n2(h) + " re f Q"); const ligne = (x1, y1, x2, y2, couleur, ep) => ec("q " + rvb(couleur) + " RG " + n2(ep || .6) + " w " + n2(x1) + " " + n2(y1) + " m " + n2(x2) + " " + n2(y2) + " l S Q"); /** Un disque, en quatre courbes — le PDF ne connaît pas le cercle. */ function disque(x, y, r, couleur){ const k = r * .5523; ec("q " + rvb(couleur) + " rg " + n2(x - r) + " " + n2(y) + " m " + n2(x - r) + " " + n2(y + k) + " " + n2(x - k) + " " + n2(y + r) + " " + n2(x) + " " + n2(y + r) + " c " + n2(x + k) + " " + n2(y + r) + " " + n2(x + r) + " " + n2(y + k) + " " + n2(x + r) + " " + n2(y) + " c " + n2(x + r) + " " + n2(y - k) + " " + n2(x + k) + " " + n2(y - r) + " " + n2(x) + " " + n2(y - r) + " c " + n2(x - k) + " " + n2(y - r) + " " + n2(x - r) + " " + n2(y - k) + " " + n2(x - r) + " " + n2(y) + " c f Q"); } /** * Un tracé du plan, reporté sur la feuille. * * Les formes viennent de la même chaîne que le SVG — « M x y L x y … Z » — * et n'emploient que des segments droits : les reprendre est une traduction, * pas un rendu. Le remplissage suit la règle pair-impair, comme le SVG, pour * que le trou d'un anneau reste un trou. */ function trace(d, vers, o){ const anneaux = anneauxDeTrace(d); if (!anneaux.length) return; let s = "q "; if (o.fond) s += rvb(o.fond) + " rg "; if (o.trait) s += rvb(o.trait) + " RG " + n2(o.ep || .4) + " w "; for (const a of anneaux){ a.forEach((pt, i) => { const q = vers(pt); s += n2(q[0]) + " " + n2(q[1]) + (i ? " l " : " m "); }); s += "h "; } ec(s + (o.fond && o.trait ? "B*" : o.fond ? "f*" : "S") + " Q"); } /** Le fichier, prêt à être enregistré. */ function sortie(titre){ /* Numérotation fixée d'avance : le catalogue doit citer les pages, et chaque page son flux, avant que l'un ou l'autre ne soit écrit. */ const CAT = 1, ARBRE = 2, F1 = 3, F2 = 4, INFO = 5; const idPage = i => INFO + 1 + i * 2, idFlux = i => INFO + 2 + i * 2; const corps = []; corps[CAT] = "<< /Type /Catalog /Pages " + ARBRE + " 0 R >>"; corps[ARBRE] = "<< /Type /Pages /Count " + pages.length + " /Kids [" + pages.map((_, i) => idPage(i) + " 0 R").join(" ") + "] >>"; const d2 = v => String(v).padStart(2, "0"); corps[INFO] = "<< /Title " + chaineUtf16(titre) + " /Producer " + chaineUtf16("Plan interactif") + " /CreationDate (D:" + d.getFullYear() + d2(d.getMonth() + 1) + d2(d.getDate()) + d2(d.getHours()) + d2(d.getMinutes()) + d2(d.getSeconds()) + ")>>"; pages.forEach((pg, i) => { corps[idPage(i)] = "<< /Type /Page /Parent " + ARBRE + " 0 R /MediaBox [0 0 " + n2(pg.l) + " " + n2(pg.h) + "] /Resources << /Font << /F1 " + F1 + " 0 R /F2 " + F2 + " 0 R >> >> /Contents " + idFlux(i) + " 0 R >>"; corps[idFlux(i)] = "<< /Length " + pg.flux.length + " >>\nstream\n" + pg.flux + "endstream"; }); /* Le fichier se compose en chaîne d'octets — un caractère par octet — puis se convertit d'un bloc : la table des positions se compte en octets depuis le début, et il faut donc écrire avant de savoir où l'on est. */ let out = "%PDF-1.4\n%\xE2\xE3\xCF\xD3\n"; for (let i = 1; i < corps.length; i++){ out += String(pos[i]).padStart(10, "0") + " 00000 n \n"; } out += "trailer\n<< /Size " + corps.length + " /Root " + CAT + " 0 R /Info " + INFO + " 0 R >>\nstartxref\n" + table + "\n%%EOF\n"; const octets = new Uint8Array(out.length); for (let i = 0; i < out.length; i++) octets[i] = out.charCodeAt(i) & 255; return octets; } return { page(paysage){ p = { l: paysage ? 841.89 : 595.28, h: paysage ? 595.28 : 841.89, flux: "" }; pages.push(p); }, vise(i){ p = pages[i]; }, get nb(){ return pages.length; }, get l(){ return p.l; }, get h(){ return p.h; }, mesure: mesure, texte: texte, coupe: coupe, rect: rect, ligne: ligne, disque: disque, trace: trace, sortie: sortie, }; } /** « M x y L x y … Z » vers une liste d'anneaux de points. */ function anneauxDeTrace(d){ const out = []; let a = null; const re = /([ML])\s*(-?[\d.]+)[\s,]+(-?[\d.]+)|(Z)/gi; let m; while ((m = re.exec(String(d || "")))){ if (m[4]){ if (a){ out.push(a); /** L'emprise d'un pavillon, recalculée si les données ne la portent pas. */ function empriseDePlan(pl){ const e = pl.emprise; if (e && isFinite(e.x0) && e.x1 > e.x0 && e.y1 > e.y0) return e; const tous = [...(pl.stands || []), ...(pl.zones || [])].filter(o => o.c && o.bb);
